//----- Broadcast received events -------------------------------------------------------------------------------------

WhenBroadcastReceived(START)
{
    Looks.SetSizeTo(55);
    Forever
    {
        Motion.GoToXY((Sensing.Of(x position, LevelDisp) - 56), (Sensing.Of(y position, LevelDisp) - 138));
        tl = 0;
        If (( Or Sensing.KeyPressed(left arrow)) Or (Sensing.MouseDown() And (Sensing.MouseX() < -25)))
        {
            Motion.TurnLeft(12);
            tl = Operator.Random(1 , 1.6);
        }
        Motion.TurnRight(4);
        If (Motion.Direction() > 90)
        {
            Motion.PointInDirection(90);
        }
        If (Motion.Direction() < 40)
        {
            Motion.PointInDirection(40);
            tl = 0;
        }
    }
}
