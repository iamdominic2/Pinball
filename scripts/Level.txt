{
    Sound.Play(Door Closing);
}

WhenBroadcastReceived(START)
{
    Life = 4;
    shield dir = 0;
    x2 dur = 0;
    Looks.SetEffectTo(GHOST, 100);
    Forever
    {
        Looks.SwitchCostumeTo((shield dir > 0));
        Motion.GoToXY(0, 0);
    }
}
