//----- Broadcast received events -------------------------------------------------------------------------------------

WhenBroadcastReceived(START)
{
    Costume.GoToFrontBack(back);
    Motion.GoToXY(25, -138);
    Looks.SetSizeTo(55);
    Forever
    {
        Costume.GoToFrontBack(back);
        Motion.GoToXY((Sensing.Of(x position, LevelDisp) + 30), (Sensing.Of(y position, LevelDisp) - 138));
        tr = 0;
        If (( Or Sensing.KeyPressed(right arrow)) Or ((Sensing.MouseDown() And (Sensing.MouseX() > 25)) And (Sensing.MouseX() < 160)))
        {
            tr = Operator.Random(1 , 1.6);
            Motion.TurnRight(12);
        }
        Motion.TurnLeft(4);
        If (Motion.Direction() > 140)
        {
            Motion.PointInDirection(140);
            tr = 0;
        }
        If (Motion.Direction() < 90)
        {
            Motion.PointInDirection(90);
        }
    }
}
