{
    NudgeX = 0;
    NudgeY = 0;
    Score = 0;
    ttt = 0;
    largedir = 0;
    Forever
    {
        NudgeX = (NudgeX / 1.5);
        NudgeY = (NudgeY / 1.5);
        ttt += 0.03;
        shield dir += -2;
        x2 dur += -2;
        largedir += -2;
        Looks.ChangeEffectBy(COLOR, 3);
    }
}
