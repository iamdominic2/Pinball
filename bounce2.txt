//----- Broadcast received events -------------------------------------------------------------------------------------

WhenBroadcastReceived(START)
{
    Looks.SwitchCostumeTo(costume1);
    Forever
    {
        Motion.SetY(0);
        Motion.SetX(206);
        Call c;
        Motion.ChangeXBy(Sensing.Of(x position, LevelDisp));
    }
}


//----- Custom blocks -------------------------------------------------------------------------------------------------

Define c (warp=true)
{
    Repeat Until Sensing.TouchingObject(bounce1)
    {
        Motion.ChangeYBy(-2);
    }
}
